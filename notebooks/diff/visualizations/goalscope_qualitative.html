def gale_shapley(men_preferences, women_preferences):
    n = len(men_preferences)
    men_free = list(range(n))
    women_free = [True] * n
    engagements = [-1] * n

    while men_free:
        man = men_free.pop(0)
        for woman in men_preferences[man]:
            if women_free[woman]:
                women_free[woman] = False
                engagements[woman] = man
                break
            else:
                current_man = engagements[woman]
                if women_preferences[woman].index(man) < women_preferences[woman].index(current_man):
                    men_free.append(current_man)
                    engagements[woman] = man
                    break

    return engagements

# Example usage:
men_preferences = [
    [1, 0, 2],
    [0, 2, 1],
    [1, 2, 0]
]

women_preferences = [
    [2, 1, 0],
    [0, 2, 1],
    [1, 0, 2]
]

print(gale_shapley(men_preferences, women_preferences))
