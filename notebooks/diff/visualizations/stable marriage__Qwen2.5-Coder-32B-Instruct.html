 def stable marriage(people):↵
    n = len(people) // 2↵
    men = people[:n]↵
    women = people[n:]↵
    men_prefs = {man: women for man in men}↵
    women_prefs = {woman: men for woman in women}↵
    engagements = {}↵
    free_men = set(men)↵
↵
    while free_men:↵
        man = free_men.pop()↵
        for woman in men_prefs[man]:↵
            if woman not in engagements:↵

                    engagements[woman] = man↵
                    free_men.add(current_man)↵
                    break↵
↵
    return engagements↵
